-- 테이블 생성
CREATE TABLE identities(
identity_id INT AUTO_INCREMENT PRIMARY KEY,
identity_name VARCHAR(20) NOT NULL,
identity_email VARCHAR(20) NOT NULL UNIQUE,
backup_email VARCHAR(20),
password VARCHAR(20) NOT NULL
);

-- 부서
create table departments(
	department_id INT AUTO_INCREMENT PRIMARY KEY,
	department_name VARCHAR(20)
);

-- 사원
CREATE TABLE employees (
    employee_id INT PRIMARY KEY,
    name VARCHAR(20) NOT NULL,
    is_admin TINYINT(1) NOT NULL DEFAULT 0,
    email VARCHAR(100) NOT NULL UNIQUE,
    phone VARCHAR(20),
    address VARCHAR(255),
    department_id INT,
    position VARCHAR(50),
    employment_type VARCHAR(20),
    hire_date DATE,
    employment_status VARCHAR(20) DEFAULT '재직',
    resignation_date DATE,
    FOREIGN KEY (employee_id) REFERENCES identities(identity_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (department_id) REFERENCES departments(department_id) ON DELETE SET NULL ON UPDATE CASCADE
);

-- 일정관리
CREATE TABLE schedules (
    schedule_id INT AUTO_INCREMENT PRIMARY KEY,
    s_content VARCHAR(255),
    employee_id INT NOT NULL,
    s_date DATE NOT NULL,
    hidden BOOLEAN DEFAULT FALSE,
    FOREIGN KEY(employee_id) REFERENCES employees(employee_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- 프로젝트
CREATE TABLE projects (
    project_id INT AUTO_INCREMENT PRIMARY KEY,
    project_name VARCHAR(100) NOT NULL,
    project_manager INT,
    project_start_date DATE NOT NULL,
    project_end_date DATE NOT NULL,
    project_progress INT DEFAULT 0,
    project_status VARCHAR(20) NOT NULL,
    FOREIGN KEY(project_manager) REFERENCES employees(employee_id) ON DELETE SET NULL ON UPDATE CASCADE
);

-- 업무
CREATE TABLE tasks(
   task_id INT AUTO_INCREMENT PRIMARY KEY,
   task_name VARCHAR(50) NOT NULL,
   task_priority VARCHAR(20),
   task_manager INT,
   task_start DATE,
   task_end DATE,
   task_progress INT,
   project_id INT,
   FOREIGN KEY(task_manager) REFERENCES employees(employee_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY(project_id) REFERENCES projects(project_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- 프로젝트 참여자
CREATE TABLE project_members (
    project_id INT NOT NULL,
    employee_id INT,
    PRIMARY KEY (project_id, employee_id),
    FOREIGN KEY(employee_id) REFERENCES employees(employee_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY(project_id) REFERENCES projects(project_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- 업무 리스트
CREATE TABLE task_checklists (
    list_number INT AUTO_INCREMENT PRIMARY KEY,
    checklist_name VARCHAR(50) NOT NULL,
    complete BOOLEAN NOT NULL DEFAULT FALSE,
    task_id INT NOT NULL,
    FOREIGN KEY (task_id) REFERENCES tasks(task_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- 고객
CREATE TABLE customers (
    customer_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_name VARCHAR(20) NOT NULL,
    customer_phone VARCHAR(20) NOT NULL
);

-- 상품
CREATE TABLE products (
    product_id INT AUTO_INCREMENT PRIMARY KEY,
    category VARCHAR(20),
    product_name VARCHAR(50) NOT NULL,
    product_description TEXT,
    product_image VARCHAR(255),
    product_price INT,
    product_description_image VARCHAR(255)
);

-- 주문
CREATE TABLE orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT,
    product_id INT,
    order_amount INT NOT NULL,
    order_status VARCHAR(20) NOT NULL DEFAULT '결제대기',
    order_date DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(product_id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- 리뷰
CREATE TABLE reviews (
    review_id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    rating TINYINT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    recommend BOOLEAN NOT NULL,
    comment TEXT,
    review_image VARCHAR(255) DEFAULT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    admin_reply TEXT DEFAULT NULL,
    replied_at DATETIME DEFAULT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(order_id) ON DELETE CASCADE ON UPDATE CASCADE
);

    inquiry_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT NOT NULL,
    employee_id INT NULL,
    inquiry_title VARCHAR(50) NOT NULL,
    inquiry_details TEXT NOT NULL,
    consultation_notes TEXT NULL,
    inquiry_status VARCHAR(20) NOT NULL,
    received_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    source_channel VARCHAR(20) NOT NULL,

    CONSTRAINT fk_inquiries_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_inquiries_employee
        FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id)
        ON DELETE SET NULL
        ON UPDATE CASCADE
);

-- 1. 월별 예산: 한 행은 '특정 월의 예산 항목 하나'
CREATE TABLE budgets (
    budget_id      INT AUTO_INCREMENT PRIMARY KEY,
    budget_name    VARCHAR(100) NOT NULL,  -- 예: 급여, 월세, 출장비, 기타
    budget_amount  BIGINT NOT NULL,        -- 편성액 (원)
    budget_month   DATE NOT NULL           -- 예: 2026년 11월 → '2026-11-01'
);

-- 2. 거래내역: 한 행은 '실제 입금 또는 지출 한 건'
CREATE TABLE financial_transactions (
    transaction_id     INT AUTO_INCREMENT PRIMARY KEY,
    transaction_title  VARCHAR(150) NOT NULL,
    transaction_type   VARCHAR(10) NOT NULL,  -- INCOME 또는 EXPENSE
    amount             BIGINT NOT NULL,
    transaction_date   DATE NOT NULL,
    memo               TEXT, 
    budget_id          INT NULL,
    project_id         INT NULL,
    created_by         INT NOT NULL,
    FOREIGN KEY (budget_id)  REFERENCES budgets(budget_id),
    FOREIGN KEY (project_id) REFERENCES projects(project_id),
    FOREIGN KEY (created_by) REFERENCES employees(employee_id)
);

-- 1. 테스트 항목과 확인 순서, 예상 결과
CREATE TABLE IF NOT EXISTS qa_test (
    test_id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    project_id INT NOT NULL,
    employee_id INT NULL,

    title VARCHAR(200) NOT NULL,
    steps TEXT NOT NULL,
    expected TEXT NOT NULL,
    
    FOREIGN KEY (project_id) REFERENCES projects (project_id) ON DELETE CASCADE,
    FOREIGN KEY (employee_id) REFERENCES employees (employee_id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 2. 테스트를 실행한 날짜와 실제 결과
CREATE TABLE IF NOT EXISTS qa_result (
    result_id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    test_id INT NOT NULL,
    employee_id INT NULL,
    
    actual TEXT NULL,
    outcome VARCHAR(10) NULL,
    executed_on DATE NULL,
    test_version INT NOT NULL DEFAULT 1,

    FOREIGN KEY (test_id) REFERENCES qa_test (test_id) ON DELETE CASCADE,
    FOREIGN KEY (employee_id) REFERENCES employees (employee_id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 3. 발견한 결함과 처리 상태
CREATE TABLE IF NOT EXISTS qa_defect (
    defect_id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    test_id INT NULL,
    employee_id INT NULL,

    title VARCHAR(200) NOT NULL,
    description TEXT NOT NULL,
    severity VARCHAR(10) NOT NULL,
    status VARCHAR(10) NOT NULL,
    resolution TEXT NULL,
    version INT NOT NULL DEFAULT 1,

    -- 프로젝트, 관련 테스트 항목, 담당 사원 연결
    FOREIGN KEY (test_id) REFERENCES qa_test (test_id) ON DELETE CASCADE,
    FOREIGN KEY (employee_id) REFERENCES employees (employee_id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
