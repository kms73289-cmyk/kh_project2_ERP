-- 근태 테이블
CREATE TABLE h_attendance (
    attendance_id INT AUTO_INCREMENT PRIMARY KEY,
    employee_id INT NOT NULL,
    attendance_date DATE NOT NULL,
    check_in TIME,
    check_out TIME,
    attendance_status VARCHAR(20) NOT NULL,
    FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id),
    UNIQUE (employee_id, attendance_date)
);

-- 연차보유현황 테이블
CREATE TABLE h_leave_balances (
    balance_id INT AUTO_INCREMENT PRIMARY KEY,
    employee_id INT NOT NULL,
    leave_year YEAR NOT NULL,
    total_days DECIMAL(4,1) NOT NULL,

    FOREIGN KEY (employee_id)
    REFERENCES employees(employee_id),

    UNIQUE (employee_id, leave_year)

-- 연차신청내역 테이블
CREATE TABLE h_leave_requests (
    request_id INT AUTO_INCREMENT PRIMARY KEY,
    employee_id INT NOT NULL,
    leave_start_date DATE NOT NULL,
    leave_end_date DATE NOT NULL,
    leave_days DECIMAL(4,1) NOT NULL,
    leave_reason VARCHAR(255),
    request_status VARCHAR(20) DEFAULT '대기',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    approved_by INT,
    processed_at DATETIME,

    FOREIGN KEY (employee_id)
    REFERENCES employees(employee_id),

    FOREIGN KEY (approved_by)
    REFERENCES employees(employee_id)
);
