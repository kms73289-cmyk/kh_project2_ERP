-- 계정
CREATE TABLE identities(
identity_id INT AUTO_INCREMENT PRIMARY KEY,
identity_name VARCHAR(20) NOT NULL,
identity_email VARCHAR(20) NOT NULL UNIQUE,
backup_email VARCHAR(20),
password VARCHAR(20) NOT NULL
);

-- 부서
create table departments(
	department_id INT AUTO_INCREMENT PRIMARY KEY,
	department_name VARCHAR(20)
);

-- 사원
CREATE TABLE employees (
    employee_id INT PRIMARY KEY,
    name VARCHAR(20) NOT NULL,
    is_admin TINYINT(1) NOT NULL DEFAULT 0,
    email VARCHAR(100) NOT NULL UNIQUE,
    phone VARCHAR(20),
    address VARCHAR(255),
    department_id INT,
    position VARCHAR(50),
    employment_type VARCHAR(20),
    hire_date DATE,
    employment_status VARCHAR(20) DEFAULT '재직',
    resignation_date DATE,
    FOREIGN KEY (employee_id) REFERENCES identities(identity_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (department_id) REFERENCES departments(department_id) ON DELETE SET NULL ON UPDATE CASCADE
);

-- 일정관리
CREATE TABLE schedules (
    schedule_id INT AUTO_INCREMENT PRIMARY KEY,
    s_content VARCHAR(255),
    employee_id INT NOT NULL,
    s_date DATE NOT NULL,
    hidden BOOLEAN DEFAULT FALSE,
    FOREIGN KEY(employee_id) REFERENCES employees(employee_id) ON DELETE CASCADE ON UPDATE CASCADE
);
